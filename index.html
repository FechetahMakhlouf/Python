05 Classes & POO

Création d'une classe

class Point:
    default_color = "red"   # attribut de CLASSE

    def __init__(self, x, y):
        self.x = x          # attribut d'INSTANCE
        self.y = y

    def draw(self):
        print(f"Point ({self.x}, {self.y})")

    @classmethod
    def zero(cls):
        return cls(0, 0)    # constructeur alternatif

point = Point(1, 2)
point.draw()
Point.default_color = "Blue"   # change pour toutes les instances

Properties & Validation

class Product:
    def __init__(self, price):
        self.price = price   # appelle le setter

    @property
    def price(self):
        return self.__price

    @price.setter
    def price(self, value):
        if value < 0:
            raise ValueError("Prix négatif impossible.")
        self.__price = value

class Stream(ABC):

Lever une exception personnalisée

class InsufficientFundsError(Exception):
    def __init__(self, amount):
        super().__init__(f"Fonds insuffisants: {amount}€")

def withdraw(amount, balance):
    if amount > balance:
        raise InsufficientFundsError(amount)
    return balance - amount

04 Views

Une view reçoit une HttpRequest et retourne une HttpResponse. C'est la couche logique.

# movies/views.py
from django.shortcuts import render, get_object_or_404
from .models import Movie

def index(request):
    movies = Movie.objects.all()
    return render(request, 'movies/index.html', {'movies': movies})

def detail(request, movie_id):
    movie = get_object_or_404(Movie, pk=movie_id)
    return render(request, "movies/detail.html", {"movie": movie})

05 Models
# movies/models.py
from django.db import models

class Genre(models.Model):
    name = models.CharField(max_length=255)

    def __str__(self):
        return self.name

class Movie(models.Model):
    title          = models.CharField(max_length=255)
    release_year   = models.IntegerField()
    number_in_stock = models.IntegerField()
    daily_rate     = models.FloatField()
    date_created   = models.DateTimeField(auto_now_add=True)
    genre          = models.ForeignKey(Genre, on_delete=models.CASCADE)

    def __str__(self):
        return self.title

movie.delete()
Movie.objects.filter(release_year__lt=2000).delete()

from tastypie.resources import ModelResource
from movies.models import Movie

class MovieResource(ModelResource):
    class Meta:
        queryset      = Movie.objects.all()
        resource_name = "movies"        # → /api/movies/
        excludes      = ["date_created"]

        fields = ['id', 'title', 'release_year', 'number_in_stock', 'daily_rate']

class MovieViewSet(viewsets.ModelViewSet):
    queryset         = Movie.objects.all()
    serializer_class = MovieSerializer
