01 Types Primitifs & Fondamentaux ▼

Python est un langage dynamiquement typé, ce qui signifie que vous n'avez pas besoin de déclarer explicitement le type d'une variable. Cependant, comprendre les types sous-jacents est crucial pour écrire du code robuste et éviter les bugs subtils liés aux conversions implicites.

Les 4 piliers de Python

Type Exemple Description détaillée Piège courant
int 42 Nombre entier de taille illimitée (pas de débordement comme en C/Java) Division / retourne toujours float
float 3.14159 Nombre décimal (IEEE 754 double precision). Précision ~15 chiffres. 0.1 + 0.2 != 0.3 → utiliser decimal pour la monnaie
bool True / False Sous-classe de int (True == 1, False == 0) if x == True: est redondant, préférer if x:

Décorateurs — modifier le comportement des fonctions

python
import time
from functools import wraps

# Décorateur simple
def timer(func):
    """Mesure le temps d'exécution d'une fonction."""
    @wraps(func)  # préserve le nom et la docstring
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        elapsed = time.time() - start
        print(f"{func.__name__} a pris {elapsed:.4f}s")
        return result
    return wrapper

@timer
def slow_function():
    time.sleep(0.1)
    return "Done"

# Décorateur avec paramètres
def repeat(times: int):
    """Exécute la fonction 'times' fois."""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            for _ in range(times):
                result = func(*args, **kwargs)
            return result
        return wrapper
    return decorator

@repeat(times=3)
def say_hello():
    print("Hello!")

say_hello()  # Affiche "Hello!" 3 fois

04 Structures de Données Maîtrisées ▼

Le choix de la structure de données détermine la performance de votre programme. Python offre des structures optimisées pour chaque cas d'usage, avec des complexités algorithmiques différentes.

list []

Tableau dynamique. Accès O(1), insertion fin O(1), insertion milieu O(n). Usage généraliste.

tuple ()

Liste immuable. Hashable (peut être clé de dict). Plus rapide et moins de mémoire que list.

set {}

Ensemble non ordonné. Test d'appartenance O(1). Supprime les doublons automatiquement.

dict {}

Table de hashage. Accès/clé O(1). Depuis Python 3.7+, conserve l'ordre d'insertion.

def fibonacci(n):
    """Génère les n premiers nombres de Fibonacci."""
    a, b = 0, 1
    for _ in range(n):
        yield a
        a, b = b, a + b

# Utilisation

# yield from — délégation de générateur
def flatten(nested_list):
    for item in nested_list:
        if isinstance(item, list):
            yield from flatten(item)  # délègue récursivement
        else:
            yield item

class Point:
    """Représente un point dans un plan 2D."""

    count = 0               # Compteur d'instances

    def zero(cls) -> "Point":
        """Constructeur alternatif — factory method."""
        return cls(0, 0)

    def from_tuple(cls, coords: tuple) -> "Point":
        """Crée un point depuis un tuple (x, y)."""
        return cls(*coords)

    @staticmethod
    def distance_formula(x1, y1, x2, y2) -> float:
        """Calcule la distance sans instance."""
        return ((x2 - x1)**2 + (y2 - y1)**2) ** 0.5

class Vector:
    def __init__(self, x: float, y: float):
        self.x = x
        self.y = y

    def __str__(self) -> str:
        return f"Vector({self.x}, {self.y})"

    def __repr__(self) -> str:
        return f"Vector({self.x!r}, {self.y!r})"

    def __eq__(self, other) -> bool:
        if not isinstance(other, Vector):
            return NotImplemented
        return self.x == other.x and self.y == other.y

    def __add__(self, other: "Vector") -> "Vector":
        return Vector(self.x + other.x, self.y + other.y)

    def __sub__(self, other: "Vector") -> "Vector":
        return Vector(self.x - other.x, self.y - other.y)

    def __mul__(self, scalar: float) -> "Vector":
        return Vector(self.x * scalar, self.y * scalar)

    def __rmul__(self, scalar: float) -> "Vector":
        return self * scalar  # 3 * v fonctionne aussi

    def __len__(self) -> int:
        return int((self.x**2 + self.y**2) ** 0.5)

    def __iter__(self):
        yield self.x
        yield self.y

v1 = Vector(2, 3)
v2 = Vector(1, 1)
print(v1 + v2)      # Vector(3, 4)
print(v1 * 3)       # Vector(6, 9)
print(3 * v1)       # Vector(6, 9) — grâce à __rmul__
print(list(v1))     # [2, 3]

class Product:
    """Produit avec validation de prix via property."""

    def __init__(self, name: str, price: float):
        self.name = name
        self._price = None      # Attribut "privé" (convention _)
        self.price = price      # Appelle le setter

    @property
    def price(self) -> float:
        """Getter — lecture contrôlée."""
        return self._price

    @price.setter
    def price(self, value: float):
        """Setter — validation à l'écriture."""
        if not isinstance(value, (int, float)):
            raise TypeError("Le prix doit être un nombre")
        if value < 0:
            raise ValueError("Le prix ne peut pas être négatif")
        self._price = float(value)

    @price.deleter
    def price(self):
        """Deleter — appelé par del obj.price"""
        print("Prix supprimé")
        self._price = None

    @property
    def discount_price(self) -> float:
        """Property calculée (read-only)."""
        return self._price * 0.9 if self._price else 0

# Utilisation

from abc import ABC, abstractmethod
from typing import List

class Stream(ABC):
    """Classe abstraite — ne peut pas être instanciée directement."""

    def __init__(self, source: str):
        self.source = source
        self.opened = False

    def open(self):
        self.opened = True

    def close(self):
        self.opened = False

    @abstractmethod
    def read(self) -> str:
        """DOIT être implémenté par les sous-classes."""
        pass

    @abstractmethod
    def write(self, data: str) -> None:
        pass

class FileStream(Stream):
    def read(self) -> str:
        return f"Lecture depuis fichier: {self.source}"

    def write(self, data: str) -> None:
        print(f"Écriture dans {self.source}: {data}")

class NetworkStream(Stream):
    def read(self) -> str:
        return f"Lecture depuis réseau: {self.source}"

    def write(self, data: str) -> None:
        print(f"Envoi sur {self.source}: {data}")

# Utilisation polymorphe
def process_stream(stream: Stream) -> None:
    """Accepte n'importe quel Stream."""
    stream.open()
    print(stream.read())
    stream.close()

# stream = Stream("test.txt")  # ❌ TypeError — classe abstraite
process_stream(FileStream("data.txt"))
process_stream(NetworkStream("http://api.com"))

06 Gestion des Exceptions & Robustesse ▼

La gestion des exceptions en Python utilise le principe EAFP (Easier to Ask Forgiveness than Permission) : il est préférable de tenter une opération et gérer l'erreur plutôt que de vérifier systématiquement avant.

Structure try-except-else-finally complète

python
def process_file(filename: str) -> dict:
    data = {}

    try:
        # Code susceptible de planter
        with open(filename, 'r') as f:
            raw = f.read()
        data = json.loads(raw)

    except FileNotFoundError as e:
        # Fichier inexistant
        print(f"Fichier non trouvé: {e.filename}")
        return {"error": "file_not_found"}

    except json.JSONDecodeError as e:
        # JSON invalide
        print(f"JSON invalide ligne {e.lineno}: {e.msg}")
        return {"error": "invalid_json"}

    except PermissionError:
        # Droits insuffisants
        print("Permission refusée")
        return {"error": "permission_denied"}

    except Exception as e:
        # Capture générique (dernier recours)
        print(f"Erreur inattendue: {e}")
        raise  # Relance pour ne pas masquer le bug

    else:
        # S'exécute UNIQUEMENT si aucune exception
        print("Traitement réussi")
        data["processed"] = True

    finally:
        # S'exécute TOUJOURS (nettoyage)
        print("Opération terminée")

    return data
ℹ️ Ordre des except : Toujours capturer les exceptions spécifiques AVANT les génériques. except Exception: avant except ValueError: rendrait cette dernière inaccessible (unreachable).

Exceptions personnalisées hiérarchiques

python
class ApplicationError(Exception):
    """Base de toutes les erreurs métier."""
    pass

class ValidationError(ApplicationError):
    """Données invalides fournies par l'utilisateur."""
    def __init__(self, field: str, message: str):
        self.field = field
        self.message = message
        super().__init__(f"[{field}] {message}")

class InsufficientFundsError(ApplicationError):
    """Solde insuffisant pour une opération."""
    def __init__(self, balance: float, amount: float):
        self.balance = balance
        self.amount = amount
        super().__init__(
            f"Fonds insuffisants: solde={balance:.2f}€, "
            f"demandé={amount:.2f}€, "
            f"manquant={amount - balance:.2f}€"
        )

class AccountFrozenError(ApplicationError):
    """Compte bloqué."""
    pass

# Utilisation
def withdraw(account, amount: float) -> float:
    if account.frozen:
        raise AccountFrozenError("Compte gelé suite à suspicion de fraude")
    if amount > account.balance:
        raise InsufficientFundsError(account.balance, amount)
    if amount <= 0:
        raise ValidationError("amount", "Le montant doit être positif")

    account.balance -= amount
    return account.balance

# Gestion hiérarchique
try:
    withdraw(user_account, 500)
except InsufficientFundsError as e:
    print(f"Transaction refusée: {e}")
    notify_user(user_account, e)
except AccountFrozenError:
    redirect_to_support()
except ApplicationError as e:
    # Capture toute erreur métier non spécifique
    log_error(e)

@contextmanager
def timer_context(label: str):
    """Mesure le temps d'un bloc de code."""
    start = time.time()

from PyPDF2 import PdfReader
from typing import List

def extract_text(file_path: str, pages: List[int] = None) -> str:
    """Extrait le texte d'un fichier PDF.

    Args:
        file_path: Chemin vers le PDF
        pages: Liste des pages à extraire (None = toutes)

    Returns:
        Texte extrait concaténé
    """
    reader = PdfReader(file_path)
    texts = []

    page_indices = pages or range(len(reader.pages))
    for i in page_indices:
        page = reader.pages[i]
        texts.append(page.extract_text())

    return "\n".join(texts)

# src/makhlouffechetah_pdf/pdf_to_image.py
from pdf2image import convert_from_path
from typing import List
from pathlib import Path

def convert_to_images(
    file_path: str,
    output_folder: str = "./",
    dpi: int = 200,
    fmt: str = "png"
) -> List[Path]:
    """Convertit un PDF en images.

    Args:
        file_path: Chemin vers le PDF
        output_folder: Dossier de sortie
        dpi: Résolution
        fmt: Format d'image (png, jpg)

    Returns:
        Liste des chemins des images créées
    """
    output_folder = Path(output_folder)
    output_folder.mkdir(parents=True, exist_ok=True)

    images = convert_from_path(file_path, dpi=dpi, fmt=fmt)
    paths = []

    for i, image in enumerate(images):
        path = output_folder / f"page_{i+1}.{fmt}"
        image.save(path)
        paths.append(path)

    return paths

from django.db import models
from django.core.validators import MinValueValidator, MaxValueValidator
from django.urls import reverse

class Genre(models.Model):
    """Catégorie de film (Action, Comédie...)."""
    name = models.CharField(max_length=100, unique=True)
    slug = models.SlugField(max_length=100, unique=True)
    description = models.TextField(blank=True)

    class Meta:
        ordering = ['name']
        verbose_name = 'Genre'
        verbose_name_plural = 'Genres'

    def __str__(self):
        return self.name

    def get_absolute_url(self):
        """URL canonique de l'objet."""
        return reverse('movies:genre-detail', kwargs={'slug': self.slug})

class Movie(models.Model):
    """Film disponible à la location."""

    # Choix prédéfinis
    RATING_CHOICES = [
        ('G', 'Tous publics'),
        ('PG', 'Accord parental'),
        ('PG-13', '-13 ans accompagné'),
        ('R', '-17 ans accompagné'),
        ('NC-17', 'Interdit -18'),
    ]

    title = models.CharField(
        max_length=255,
        db_index=True,  # Index pour recherche rapide
        help_text="Titre original du film"
    )
    slug = models.SlugField(max_length=255, unique=True)

    # Relations
    genre = models.ForeignKey(
        Genre,
        on_delete=models.PROTECT,  # Empêche suppression si films existants
        related_name='movies',     # genre.movies retourne tous les films
        null=True,
        blank=True,
    )

    # Champs numériques avec validation
    release_year = models.IntegerField(
        validators=[MinValueValidator(1900), MaxValueValidator(2030)],
        help_text="Année de sortie"
    )
    number_in_stock = models.PositiveIntegerField(default=0)
    daily_rate = models.DecimalField(
        max_digits=5, decimal_places=2,
        validators=[MinValueValidator(0)],
    )
    rating = models.CharField(
        max_length=5,
        choices=RATING_CHOICES,
        default='G',
    )

    # Champs booléens et temporels
    is_active = models.BooleanField(default=True)
    date_created = models.DateTimeField(auto_now_add=True)
    date_updated = models.DateTimeField(auto_now=True)

    # Média
    poster = models.ImageField(
        upload_to='movies/posters/%Y/%m/',
        blank=True,
        null=True,
    )

    class Meta:
        ordering = ['-date_created']
        verbose_name = 'Film'
        verbose_name_plural = 'Films'
        indexes = [
            models.Index(fields=['genre', 'is_active']),
        ]
        constraints = [
            models.CheckConstraint(
                check=models.Q(release_year__gte=1900),
                name='valid_release_year',
            ),
        ]

    def __str__(self):
        return f"{self.title} ({self.release_year})"

    def get_absolute_url(self):
        return reverse('movies:detail', kwargs={'movie_id': self.id})

    @property
    def is_available(self):
        """Propriété calculée — pas de colonne en BDD."""
        return self.is_active and self.number_in_stock > 0

    def rent(self):
        """Méthode métier — décrémente le stock."""
        if self.number_in_stock <= 0:
            raise ValueError("Film non disponible")
        self.number_in_stock -= 1
        self.save(update_fields=['number_in_stock'])

    def return_copy(self):
        """Méthode métier — incrémente le stock."""
        self.number_in_stock += 1
        self.save(update_fields=['number_in_stock'])

# ❌ PROBLÈME N+1 : 1 requête pour les films + N requêtes pour les genres
for movie in Movie.objects.all():
    print(movie.genre.name)  # Requête BDD à chaque itération !

# ✅ SOLUTION 1 : select_related (JOIN SQL pour ForeignKey/OneToOne)
for movie in Movie.objects.select_related('genre').all():
    print(movie.genre.name)  # Aucune requête supplémentaire

# ✅ SOLUTION 2 : prefetch_related (2 requêtes pour ManyToMany/Reverse FK)
for genre in Genre.objects.prefetch_related('movies').all():
    for movie in genre.movies.all():
        print(movie.title)

# ✅ SOLUTION 3 : values_list (récupère seulement les colonnes nécessaires)
titles = Movie.objects.values_list('title', flat=True)
# flat=True pour une liste simple, sinon tuples

# ✅ SOLUTION 4 : only / defer (chargement partiel)
movies = Movie.objects.only('title', 'release_year')  # Charge seulement ces champs
movies = Movie.objects.defer('poster')  # Exclut le champ lourd

# Annotations et agrégations

# Annotation : ajoute un champ calculé à chaque objet
movies = Movie.objects.annotate(
    rental_count=Count('rentals'),

)

stats = Movie.objects.aggregate(
    total=Count('id'),
    avg_rate=Avg('daily_rate'),
    max_year=Max('release_year'),
    min_year=Min('release_year'),
    total_stock=Sum('number_in_stock')

@register.filter
def star_rating(value):
    """Convertit une note 0-5 en étoiles HTML."""
    full = int(value)
    empty = 5 - full
    stars = '★' * full + '☆' * empty
    return mark_safe(f'<span class="text-warning">{stars}</span>')

from rest_framework.response import Response
from django_filters.rest_framework import DjangoFilterBackend
from movies.models import Movie
from .serializers import MovieSerializer

class MovieViewSet(viewsets.ModelViewSet):
    """
    ViewSet complet : list, create, retrieve, update, partial_update, destroy.
    """
    queryset = Movie.objects.all().select_related('genre')
    serializer_class = MovieSerializer
    filter_backends = [DjangoFilterBackend, filters.SearchFilter, filters.OrderingFilter]
    filterset_fields = ['genre', 'is_active', 'rating', 'release_year']
    search_fields = ['title', 'genre__name']
    ordering_fields = ['title', 'release_year', 'daily_rate', 'date_created']
    ordering = ['-date_created']

    @action(detail=False, methods=['get'])
    def available(self, request):
        """Endpoint personnalisé : /api/movies/available/"""
        movies = self.get_queryset().filter(is_active=True, number_in_stock__gt=0)
        serializer = self.get_serializer(movies, many=True)
        return Response(serializer.data)

    @action(detail=True, methods=['post'])
    def rent(self, request, pk=None):
        """Endpoint personnalisé : /api/movies/1/rent/"""
        movie = self.get_object()
        try:
            movie.rent()
            return Response({'status': 'film loué', 'stock': movie.number_in_stock})
        except ValueError as e:
            return Response({'error': str(e)}, status=400)
